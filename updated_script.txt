import json
import pickle
import requests
from sklearn.metrics import classification_report, accuracy_score

# -----------------------------
# Step 1: Load old encoder
# -----------------------------
encoder_path = "path_to_encoder_2800.pkl"  # Replace with your encoder path
with open(encoder_path, "rb") as f:
    old_encoder = pickle.load(f)

known_labels = old_encoder.classes_
print(f"Model can predict {len(known_labels)} labels.")

# -----------------------------
# Step 2: Load your Label Studio dataset using your function
# -----------------------------
def load_labelstudio(path):
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)

    texts = []
    labels = []

    for task in data:
        text = task["data"].get("html") or task["data"].get("text")  # fallback if key differs

        for ann in task.get("annotations", []):
            for res in ann.get("result", []):
                if res.get("type") == "taxonomy":
                    cats = res["value"].get("taxonomy", [])
                    if isinstance(cats, list) and len(cats) > 0:
                        full_label = " > ".join(cats)  # FULL PATH
                        texts.append(text)
                        labels.append(full_label)

    return texts, labels

dataset_path = "path_to_5200_dataset.json"  # Replace with your dataset path
texts, true_labels_raw = load_labelstudio(dataset_path)

# -----------------------------
# Step 3: Filter true labels to known labels
# -----------------------------
def map_to_known_label(label, known_labels):
    return label if label in known_labels else "UNK"

true_labels = [map_to_known_label(lbl, known_labels) for lbl in true_labels_raw]

# -----------------------------
# Step 4: Batch ML backend predictions (safe for prod)
# -----------------------------
ml_backend_url = "http://localhost:9090/predict"  # Replace with your backend URL
batch_size = 50  # Adjust depending on backend load

def ml_backend_predict_batch(texts, url=ml_backend_url, batch_size=50):
    predictions = []
    for i in range(0, len(texts), batch_size):
        batch_texts = texts[i:i+batch_size]
        payload = {"data": [{"text": txt} for txt in batch_texts]}
        try:
            response = requests.post(url, json=payload)
            response_json = response.json()
            # Adjust depending on your backend response structure
            for pred in response_json['predictions']:
                label = pred if pred in known_labels else "UNK"
                predictions.append(label)
        except Exception as e:
            print(f"Batch prediction error at index {i}-{i+len(batch_texts)-1}: {e}")
            predictions.extend(["UNK"] * len(batch_texts))
    return predictions

pred_labels = ml_backend_predict_batch(texts, batch_size=batch_size)

# -----------------------------
# Step 5: Compute evaluation metrics
# -----------------------------
print("Accuracy:", accuracy_score(true_labels, pred_labels))
print("\nClassification Report:")
print(classification_report(true_labels, pred_labels, labels=list(known_labels)+["UNK"]))
