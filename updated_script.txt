import json, os, pickle, numpy as np, pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import classification_report, accuracy_score, confusion_matrix
from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

# ================================
# CHANGE ONLY THESE 3 PATHS
# ================================
JSON_PATH = "project.json"                     # your exported labelstudio JSON
MODEL_DIR = "model_folder"                     # your fine-tuned BERT model
ENCODER_FILE = "encoders/label_encoder.pkl"    # old encoder used during training
# ================================

BATCH_SIZE = 16
MAX_EXAMPLES_TO_SHOW = 20


# -----------------------------------------------------------
# 1. LOAD LABEL STUDIO DATA (full taxonomy hierarchy)
# -----------------------------------------------------------
def load_data(path):
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)

    texts = []
    labels = []

    for task in data:
        text = task["data"].get("html") or task["data"].get("text") or ""
        full_label = None

        for ann in task.get("annotations", []):
            for res in ann.get("result", []):
                if res.get("type") == "taxonomy":
                    cats = res["value"].get("taxonomy", [])
                    clean = []

                    for c in cats:
                        if isinstance(c, list):
                            for x in c:
                                if isinstance(x, str):
                                    clean.append(x)
                        elif isinstance(c, str):
                            clean.append(c)

                    if clean:
                        full_label = " > ".join(clean)
                        break
            if full_label:
                break

        if full_label:
            texts.append(text)
            labels.append(full_label)

    return texts, labels


print("\nLoading dataset...")
texts, labels = load_data(JSON_PATH)
print("Total samples loaded:", len(texts))


# -----------------------------------------------------------
# 2. LOAD MODEL + OLD ENCODER
# -----------------------------------------------------------
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using device:", device)

tokenizer = AutoTokenizer.from_pretrained(MODEL_DIR)
model = AutoModelForSequenceClassification.from_pretrained(MODEL_DIR).to(device)

with open(ENCODER_FILE, "rb") as f:
    le = pickle.load(f)

print("\nModel num_labels:", model.config.num_labels)
print("Encoder label count:", len(le.classes_))


# -----------------------------------------------------------
# 3. FILTER OUT LABELS NOT PRESENT IN OLD ENCODER
# -----------------------------------------------------------
print("\nFiltering unseen labels...")

valid_labels = set(le.classes_)
filtered_texts = []
filtered_labels = []

for t, lbl in zip(texts, labels):
    if lbl in valid_labels:
        filtered_texts.append(t)
        filtered_labels.append(lbl)

print("Before filtering:", len(texts))
print("After filtering :", len(filtered_texts))
print("Dropped        :", len(texts) - len(filtered_texts))

texts = filtered_texts
labels = filtered_labels


# -----------------------------------------------------------
# 4. ENCODE LABELS + TRAIN/TEST SPLIT
# -----------------------------------------------------------
y = le.transform(labels)

X_train, X_test, y_train, y_test = train_test_split(
    texts, y, test_size=0.20, random_state=42, stratify=y
)

print("\nTrain size:", len(X_train))
print("Test size :", len(X_test))

# Test label distribution
unique, counts = np.unique(y_test, return_counts=True)
dist = sorted(zip(unique, counts), key=lambda x: -x[1])

print("\nTop label counts in test set:")
for idx, count in dist[:20]:
    print(f"{idx} : {count} : {le.classes_[idx]}")


# -----------------------------------------------------------
# 5. BATCH PREDICTION FUNCTION (Softmax)
# -----------------------------------------------------------
def batch_predict(texts):
    preds_idx = []
    preds_conf = []
    model.eval()

    with torch.no_grad():
        for i in range(0, len(texts), BATCH_SIZE):
            batch = texts[i:i+BATCH_SIZE]
            enc = tokenizer(batch, truncation=True, padding=True,
                            return_tensors="pt").to(device)

            out = model(**enc)
            logits = out.logits.cpu().numpy()

            # numerical stable softmax
            exp = np.exp(logits - np.max(logits, axis=1, keepdims=True))
            probs = exp / exp.sum(axis=1, keepdims=True)

            p = probs.argmax(axis=1)
            c = probs[np.arange(len(p)), p]

            preds_idx.extend(p.tolist())
            preds_conf.extend(c.tolist())

    return np.array(preds_idx), np.array(preds_conf)


print("\nRunning predictions...")
pred_idx, pred_conf = batch_predict(X_test)

print("Finished predictions.")


# -----------------------------------------------------------
# 6. EVALUATION METRICS
# -----------------------------------------------------------
acc = accuracy_score(y_test, pred_idx)
print("\n============================")
print(" FINAL ACCURACY :", acc)
print("============================\n")

print("Classification Report:")
print(classification_report(
    y_test,
    pred_idx,
    target_names=le.classes_,
    zero_division=0
))


# -----------------------------------------------------------
# 7. SAMPLE PREDICTION TABLE FOR AUDIT
# -----------------------------------------------------------
rows = []

for i in range(min(MAX_EXAMPLES_TO_SHOW, len(X_test))):
    rows.append({
        "text_snippet": X_test[i][:200].replace("\n", " "),
        "true_label": le.inverse_transform([y_test[i]])[0],
        "pred_label": le.inverse_transform([pred_idx[i]])[0],
        "confidence": float(pred_conf[i])
    })

df_sample = pd.DataFrame(rows)
print("\nSample evaluation rows:")
print(df_sample.to_string(index=False))


# -----------------------------------------------------------
# 8. SAVE FULL CSV FOR AUDIT TEAM
# -----------------------------------------------------------
out_df = pd.DataFrame({
    "text": X_test,
    "true_label": [le.inverse_transform([i])[0] for i in y_test],
    "pred_label": [le.inverse_transform([i])[0] for i in pred_idx],
    "confidence": pred_conf
})

out_df.to_csv("evaluation_results.csv", index=False)
print("\nSaved: evaluation_results.csv")
print("Done.")
