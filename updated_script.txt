# ---------- 1) load and extract robust full-path labels ----------
def load_data(path):
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    texts, labels = [], []
    for task in data:
        text = task["data"].get("html") or task["data"].get("text") or ""
        # find first taxonomy result
        full_label = None
        for ann in task.get("annotations", []):
            for res in ann.get("result", []):
                if res.get("type") == "taxonomy':
                    cats = res["value"].get("taxonomy", [])
                    # make list of strings (flatten nested lists and drop non-strings)
                    clean = []
                    for c in cats:
                        if isinstance(c, list):
                            for x in c:
                                if isinstance(x, str):
                                    clean.append(x)
                        elif isinstance(c, str):
                            clean.append(c)
                    if clean:
                        full_label = " > ".join(clean)
                        break
            if full_label:
                break
        if full_label:
            texts.append(text)
            labels.append(full_label)
    return texts, labels

# ---------- 4) batch-predict the whole X_test (collect logits for confidence) ----------
def batch_predict(texts, tokenizer, model, batch_size=16, device=device):
    model.eval()
    preds_idx = []
    preds_probs = []
    with torch.no_grad():
        for i in range(0, len(texts), batch_size):
            batch_texts = texts[i:i+batch_size]
            enc = tokenizer(batch_texts, truncation=True, padding=True, return_tensors="pt").to(device)
            out = model(**enc)
            logits = out.logits.cpu().numpy()
            # If model was trained single-label, use softmax; else if multi-label it would use sigmoid
            probs = np.exp(logits - np.max(logits, axis=1, keepdims=True))
            probs = probs / probs.sum(axis=1, keepdims=True)   # softmax numerically stable
            pred = probs.argmax(axis=1)
            preds_idx.extend(pred.tolist())
            # keep top prob per example
            top_probs = probs[np.arange(len(pred)), pred]
            preds_probs.extend(top_probs.tolist())
    return np.array(preds_idx), np.array(preds_probs)
