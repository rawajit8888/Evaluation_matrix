def prepare_data(df, label_column="taxonomy", random_state=42):
    """
    Performs safe train-test split ensuring all labels are included.
    If a label has only 1 sample → go to TRAIN.
    Otherwise → 1 sample goes to TEST, rest to TRAIN.
    """

    test_rows = []
    train_rows = []

    for label, group in df.groupby(label_column):
        if len(group) == 1:
            # Only 1 sample → keep in TRAIN only
            train_rows.append(group)
        else:
            # 1 sample forced into TEST → rest to TRAIN
            test_sample = group.sample(1, random_state=random_state)
